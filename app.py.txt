# Definición exacta de la arquitectura ganadora (Ítem 3.3: Híbrido LSTM + MLP)
class HybridLSTMNet(nn.Module):
    def __init__(self, seq_dim, stat_dim):
        super().__init__()
        self.lstm = nn.LSTM(seq_dim, 24, batch_first=True)
        self.stat_net = nn.Sequential(
            nn.Linear(stat_dim, 48),
            nn.BatchNorm1d(48),
            nn.ReLU(),
            nn.Dropout(0.4)
        )
        self.head = nn.Sequential(
            nn.Linear(24 + 48, 32),
            nn.BatchNorm1d(32),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(32, 3)
        )

    def forward(self, x_seq, x_stat):
        _, (h_n, _) = self.lstm(x_seq)
        return self.head(torch.cat([h_n[-1], self.stat_net(x_stat)], dim=1))

@st.cache_resource
def cargar_recursos():
    art = joblib.load("artefactos_hibrido.joblib")
    model = HybridLSTMNet(seq_dim=6, stat_dim=art["stat_dim"])
    model.load_state_dict(torch.load("modelo_hibrido_lstm.pt", map_location=torch.device("cpu")))
    model.eval()
    return model, art

def predecir_estudiante(df_fila, model, art):
    s1 = art["scaler_sem"].transform(df_fila[art["sem1_cols"]].values)
    s2 = art["scaler_sem"].transform(df_fila[art["sem2_cols"]].values)
    x_seq = torch.tensor(np.stack([s1, s2], axis=1), dtype=torch.float32)
    x_stat = torch.tensor(art["prep_stat"].transform(df_fila[art["static_features"]]), dtype=torch.float32)

    with torch.no_grad():
        logits = model(x_seq, x_stat)
        probs = torch.softmax(logits, dim=1).numpy()[0]

    clases = art["label_encoder"].classes_
    pred_idx = int(np.argmax(probs))
    return clases[pred_idx], dict(zip(clases, probs))

# Cargar modelo y pipeline
